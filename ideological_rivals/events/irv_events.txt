namespace = irv

# --- Setup ------------------------------------------------------------------------------------

# Game start, once, for the first human player: choose 1-3 rivals by galaxy size
# (tiny, small: 1; medium, large: 2; huge and larger: 3).
country_event = {
	id = irv.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		is_ai = no
		NOT = { has_global_flag = irv_setup_done }
	}
	immediate = {
		set_global_flag = irv_setup_done
		save_global_event_target_as = irv_player
		capital_scope = { save_global_event_target_as = irv_cap_0 }
		irv_init_reachable = yes
		irv_choose_rival = { N = 1 }
		if = {
			limit = { NOR = { galaxy_size = tiny galaxy_size = small } }
			irv_choose_rival = { N = 2 }
		}
		if = {
			limit = { NOR = { galaxy_size = tiny galaxy_size = small galaxy_size = medium galaxy_size = large } }
			irv_choose_rival = { N = 3 }
		}
		every_country = {
			limit = { has_country_flag = irv_rival }
			add_modifier = { modifier = irv_rival_drive }
			add_opinion_modifier = { who = event_target:irv_player modifier = irv_opinion_rival }
		}
		if = {
			limit = { any_country = { has_country_flag = irv_rival } }
			country_event = { id = irv.2 days = 30 }
		}
	}
}

# The player learns that rivals exist (no names: they have not met yet).
country_event = {
	id = irv.2
	title = irv.2.name
	desc = irv.2.desc
	picture = GFX_evt_raised_fist
	is_triggered_only = yes
	option = { name = irv.2.a }
}

# --- Monthly, rivals only ---------------------------------------------------------------------

# Found the ideology as soon as the rival knows another empire (vanilla needs a contact), and
# feed willpower once the player is on Force of Will too.
country_event = {
	id = irv.10
	hide_window = yes
	is_triggered_only = yes
	trigger = { has_country_flag = irv_rival }
	immediate = {
		if = {
			limit = { irv_rival_needs_ideology = yes }
			irv_found_ideology = yes
		}
		irv_grant_ambition = yes
		if = {
			limit = {
				has_global_flag = irv_player_has_force_of_will
				has_ascension_perk = ap_force_of_will
			}
			add_resource = { willpower = @irv_willpower_monthly }
		}
	}
}

# --- Ascension perks --------------------------------------------------------------------------

# Someone picked a perk. The player on Force of Will: the rivals redouble their efforts.
# A rival: its ambition may be due now (3 perks), before its next pick could be another ambition.
# If its AI took its ambition itself, it is announced here.
country_event = {
	id = irv.20
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		OR = {
			has_country_flag = irv_rival
			AND = {
				is_ai = no
				has_ascension_perk = ap_force_of_will
				NOT = { has_global_flag = irv_player_has_force_of_will }
			}
		}
	}
	immediate = {
		if = {
			limit = { has_country_flag = irv_rival }
			irv_grant_ambition = yes
			if = {
				limit = { has_ambition = yes }
				irv_announce = yes
			}
		}
		else = {
			set_global_flag = irv_player_has_force_of_will
			every_country = {
				limit = { has_country_flag = irv_rival }
				add_modifier = { modifier = irv_rival_redoubled }
			}
			if = {
				limit = { any_country = { has_country_flag = irv_rival } }
				country_event = { id = irv.21 }
			}
		}
	}
}

# --- Console test helpers ---------------------------------------------------------------------

# event irv.900: list the rivals, their chosen ethics, aim, ideology and ambition in game.log.
# Every line names its rival: game.log drops a line identical to an earlier one.
country_event = {
	id = irv.900
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		log = "IRV player: [This.GetName]"
		if = {
			limit = { exists = event_target:irv_rival_1 }
			event_target:irv_rival_1 = {
				log = "IRV rival 1: [This.GetName], capital [This.Capital.GetName]"
				irv_log_rival = { N = 1 }
			}
		}
		if = {
			limit = { exists = event_target:irv_rival_2 }
			event_target:irv_rival_2 = {
				log = "IRV rival 2: [This.GetName], capital [This.Capital.GetName]"
				irv_log_rival = { N = 2 }
			}
		}
		if = {
			limit = { exists = event_target:irv_rival_3 }
			event_target:irv_rival_3 = {
				log = "IRV rival 3: [This.GetName], capital [This.Capital.GetName]"
				irv_log_rival = { N = 3 }
			}
		}
	}
}

# event irv.902 (test): every rival gets 3 ascension perks; irv.10 grants its ambition next month.
country_event = {
	id = irv.902
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_country = {
			limit = { has_country_flag = irv_rival }
			log = "IRV test: 3 perks for [This.GetName]"
			if = {
				limit = { NOT = { has_ascension_perk = ap_technological_ascendancy } }
				add_ascension_perk = ap_technological_ascendancy
			}
			if = {
				limit = { NOT = { has_ascension_perk = ap_one_vision } }
				add_ascension_perk = ap_one_vision
			}
			if = {
				limit = { NOT = { has_ascension_perk = ap_interstellar_dominion } }
				add_ascension_perk = ap_interstellar_dominion
			}
		}
	}
}

# event irv.903 (test): every empire's spread score and its distances to the player's capital,
# to check which measure `distance` uses by default and that the farthest candidate was chosen.
country_event = {
	id = irv.903
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_country = {
			limit = { is_country_type = default exists = capital_scope }
			set_variable = { which = irv_dbg_spread value = value:irv_spread_1 }
			set_variable = { which = irv_dbg_euclid value = value:irv_dbg_distance|TYPE|euclidean| }
			set_variable = { which = irv_dbg_lanes value = value:irv_dbg_distance|TYPE|hyperlane| }
			log = "IRV spread [This.Capital.GetName]: spread [This.irv_dbg_spread] euclid [This.irv_dbg_euclid] lanes [This.irv_dbg_lanes]"
			if = {
				limit = { irv_can_be_rival = yes }
				log = "IRV   candidate: [This.Capital.GetName]"
			}
			if = {
				limit = { has_country_flag = irv_rival }
				log = "IRV   rival: [This.Capital.GetName]"
			}
		}
	}
}
