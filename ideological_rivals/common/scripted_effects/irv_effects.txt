# Ideological Rivals: scripted effects
#
# Ethic axes, each with a global flag irv_axis_<axis> once a rival has taken it as its primary:
#   authority  egalitarian / authoritarian
#   xeno       xenophile / xenophobe
#   war        pacifist / militarist
#   spirit     spiritualist / materialist
# event_target:irv_player is the human player the rivals are chosen against.

# --- Choosing a rival -------------------------------------------------------------------------

# Make the best-spread candidate rival number $N$. Scope: none (global). Needs irv_cap_0..N-1.
irv_choose_rival = {
	ordered_country = {
		limit = { irv_can_be_rival = yes }
		order_by = value:irv_spread_$N$
		position = 0
		set_country_flag = irv_rival
		save_global_event_target_as = irv_rival_$N$
		capital_scope = { save_global_event_target_as = irv_cap_$N$ }
		irv_shape_rival = yes
	}
}

# --- Ethics -----------------------------------------------------------------------------------

# Primary: the opposite of one of the player's ethics whose axis no rival has taken yet.
# Scope: rival. PLAYER = the player's ethic key, AXIS = its axis, OPP = the opposite ethic.
irv_try_primary = {
	if = {
		limit = {
			NOT = { has_country_flag = irv_primary_done }
			NOT = { has_global_flag = irv_axis_$AXIS$ }
			event_target:irv_player = { has_ethic = $PLAYER$ }
		}
		set_country_flag = irv_primary_done
		set_country_flag = irv_axis_$AXIS$
		set_country_flag = irv_primary_$OPP$
		set_global_flag = irv_axis_$AXIS$
	}
}

# Secondary: the opposite of another of the player's ethics, on an axis other than the primary.
# Scope: rival. ETHIC = the player's ethic (without ethic_ and fanatic_), OPP = its opposite.
irv_try_secondary = {
	if = {
		limit = {
			NOT = { has_country_flag = irv_secondary_done }
			NOT = { has_country_flag = irv_axis_$AXIS$ }
			event_target:irv_player = {
				OR = {
					has_ethic = ethic_$ETHIC$
					has_ethic = ethic_fanatic_$ETHIC$
				}
			}
		}
		set_country_flag = irv_secondary_done
		set_country_flag = irv_secondary_$OPP$
	}
}

# Fallback when the player offers no axis left (gestalt player, or more rivals than player axes).
# Scope: rival. KIND = primary or secondary; a primary also claims the axis globally. Never an
# ethic the player holds, never an authoritarian primary (irv_ethic_barred_*).
irv_random_ethic = {
	random_list = {
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = authority } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = egalitarian } }
			irv_set_random_$KIND$ = { AXIS = authority E = egalitarian }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = authority } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = authoritarian } }
			modifier = { factor = 0 irv_authoritarian_barred_$KIND$ = { E = authoritarian } }
			irv_set_random_$KIND$ = { AXIS = authority E = authoritarian }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = xeno } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = xenophile } }
			irv_set_random_$KIND$ = { AXIS = xeno E = xenophile }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = xeno } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = xenophobe } }
			irv_set_random_$KIND$ = { AXIS = xeno E = xenophobe }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = war } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = pacifist } }
			irv_set_random_$KIND$ = { AXIS = war E = pacifist }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = war } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = militarist } }
			irv_set_random_$KIND$ = { AXIS = war E = militarist }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = spirit } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = spiritualist } }
			irv_set_random_$KIND$ = { AXIS = spirit E = spiritualist }
		}
		1 = {
			modifier = { factor = 0 irv_axis_blocked_$KIND$ = { AXIS = spirit } }
			modifier = { factor = 0 irv_ethic_barred_$KIND$ = { E = materialist } }
			irv_set_random_$KIND$ = { AXIS = spirit E = materialist }
		}
	}
}

irv_set_random_primary = {
	set_country_flag = irv_primary_done
	set_country_flag = irv_primary_$E$
	set_country_flag = irv_axis_$AXIS$
	set_global_flag = irv_axis_$AXIS$
}

irv_set_random_secondary = {
	set_country_flag = irv_secondary_done
	set_country_flag = irv_secondary_$E$
	set_country_flag = irv_secondary_axis_$AXIS$
}

# Replace the ethos by the chosen ethics and repair the government. Scope: rival.
irv_apply_ethics = {
	clear_ethos = yes
	irv_add_ethic = { E = egalitarian }
	irv_add_ethic = { E = authoritarian }
	irv_add_ethic = { E = xenophile }
	irv_add_ethic = { E = xenophobe }
	irv_add_ethic = { E = pacifist }
	irv_add_ethic = { E = militarist }
	irv_add_ethic = { E = spiritualist }
	irv_add_ethic = { E = materialist }
	irv_fix_authority = yes
}

irv_add_ethic = {
	if = {
		limit = { has_country_flag = irv_primary_$E$ }
		country_add_ethic = ethic_fanatic_$E$
		every_owned_pop_group = {
			pop_change_ethic = { ethic = ethic_$E$ percentage = @irv_pop_ethic_share }
		}
	}
	if = {
		limit = { has_country_flag = irv_secondary_$E$ }
		country_add_ethic = ethic_$E$
	}
}

# Authorities that the new ethics forbid (common/governments/authorities/00_authorities.txt):
# democratic: no authoritarian; oligarchic: no fanatic egalitarian/authoritarian;
# dictatorial and imperial: no egalitarian.
irv_fix_authority = {
	if = {
		limit = {
			has_ethic = ethic_fanatic_authoritarian
			OR = { has_authority = auth_democratic has_authority = auth_oligarchic }
		}
		change_government = { authority = auth_dictatorial remove_invalid_civics = yes cooldown = no }
	}
	else_if = {
		limit = {
			has_ethic = ethic_fanatic_egalitarian
			OR = { has_authority = auth_dictatorial has_authority = auth_imperial has_authority = auth_oligarchic }
		}
		change_government = { authority = auth_democratic remove_invalid_civics = yes cooldown = no }
	}
	else_if = {
		limit = { has_ethic = ethic_authoritarian has_authority = auth_democratic }
		change_government = { authority = auth_oligarchic remove_invalid_civics = yes cooldown = no }
	}
	else_if = {
		limit = {
			has_ethic = ethic_egalitarian
			OR = { has_authority = auth_dictatorial has_authority = auth_imperial }
		}
		change_government = { authority = auth_oligarchic remove_invalid_civics = yes cooldown = no }
	}
	else = {
		change_government = { remove_invalid_civics = yes cooldown = no }
	}
}

# Pick and apply a rival's ethics, then its ambition. Scope: rival. The player's fanatic ethic is
# used first. Authoritarian is never a primary: vanilla Force of Will bars an authoritarian
# ideology, and no other ambition suits it, so an egalitarian player is opposed by secondaries.
irv_shape_rival = {
	irv_try_primary = { PLAYER = ethic_fanatic_authoritarian AXIS = authority OPP = egalitarian }
	irv_try_primary = { PLAYER = ethic_fanatic_xenophile AXIS = xeno OPP = xenophobe }
	irv_try_primary = { PLAYER = ethic_fanatic_xenophobe AXIS = xeno OPP = xenophile }
	irv_try_primary = { PLAYER = ethic_fanatic_pacifist AXIS = war OPP = militarist }
	irv_try_primary = { PLAYER = ethic_fanatic_militarist AXIS = war OPP = pacifist }
	irv_try_primary = { PLAYER = ethic_fanatic_spiritualist AXIS = spirit OPP = materialist }
	irv_try_primary = { PLAYER = ethic_fanatic_materialist AXIS = spirit OPP = spiritualist }
	irv_try_primary = { PLAYER = ethic_authoritarian AXIS = authority OPP = egalitarian }
	irv_try_primary = { PLAYER = ethic_xenophile AXIS = xeno OPP = xenophobe }
	irv_try_primary = { PLAYER = ethic_xenophobe AXIS = xeno OPP = xenophile }
	irv_try_primary = { PLAYER = ethic_pacifist AXIS = war OPP = militarist }
	irv_try_primary = { PLAYER = ethic_militarist AXIS = war OPP = pacifist }
	irv_try_primary = { PLAYER = ethic_spiritualist AXIS = spirit OPP = materialist }
	irv_try_primary = { PLAYER = ethic_materialist AXIS = spirit OPP = spiritualist }
	if = {
		limit = { NOT = { has_country_flag = irv_primary_done } }
		irv_random_ethic = { KIND = primary }
	}
	irv_try_secondary = { ETHIC = egalitarian AXIS = authority OPP = authoritarian }
	irv_try_secondary = { ETHIC = authoritarian AXIS = authority OPP = egalitarian }
	irv_try_secondary = { ETHIC = xenophile AXIS = xeno OPP = xenophobe }
	irv_try_secondary = { ETHIC = xenophobe AXIS = xeno OPP = xenophile }
	irv_try_secondary = { ETHIC = pacifist AXIS = war OPP = militarist }
	irv_try_secondary = { ETHIC = militarist AXIS = war OPP = pacifist }
	irv_try_secondary = { ETHIC = spiritualist AXIS = spirit OPP = materialist }
	irv_try_secondary = { ETHIC = materialist AXIS = spirit OPP = spiritualist }
	if = {
		limit = { NOT = { has_country_flag = irv_secondary_done } }
		irv_random_ethic = { KIND = secondary }
	}
	irv_apply_ethics = yes
	irv_choose_aim = yes
}

# --- Ambition ---------------------------------------------------------------------------------

# Which ambition the rival will pursue. Scope: rival, after its ethics are set.
# Militarist: Become the Crisis; xenophobe: Galactic Hyperthermia (vanilla bars both from
# xenophiles and pacifists, and its AI takes them only as militarist or xenophobe); any other
# primary: Force of Will. Without the crisis's DLC, or on Civilian difficulty (vanilla
# blocks_disruptive_gameplay), a crisis rival pursues none and stays a rival only.
irv_choose_aim = {
	if = {
		limit = { has_country_flag = irv_primary_militarist }
		if = {
			limit = { has_nemesis = yes blocks_disruptive_gameplay = no }
			set_country_flag = irv_aim_crisis
		}
	}
	else_if = {
		limit = { has_country_flag = irv_primary_xenophobe }
		if = {
			limit = { has_infernals = yes blocks_disruptive_gameplay = no }
			set_country_flag = irv_aim_hyperthermia
		}
	}
	else = {
		set_country_flag = irv_aim_fow
	}
}

# Grant the rival its ambition once it holds 3 perks (every ambition's own threshold). Scope: rival.
# The AI would rarely take a crisis by itself (vanilla ai_weight: 0 below 10 % of the galaxy, 0
# for a non-infernal Hyperthermia), so the mod adds the perk. add_ascension_perk skips `possible`,
# so a Force of Will rival is checked against vanilla's own conditions here.
irv_grant_ambition = {
	if = {
		limit = {
			has_ambition = no
			num_ascension_perks > 2
			is_subject = no
		}
		if = {
			limit = { has_country_flag = irv_aim_crisis }
			add_ascension_perk = ap_become_the_crisis
		}
		else_if = {
			limit = { has_country_flag = irv_aim_hyperthermia }
			add_ascension_perk = ap_galactic_hyperthermia
		}
		else_if = {
			limit = {
				has_country_flag = irv_aim_fow
				is_fanatic = yes
				exists = ideology
				ideology = { is_ideology_custodian = root }
			}
			add_ascension_perk = ap_force_of_will
		}
		if = {
			limit = { has_ambition = yes }
			log = "IRV ambition granted: [This.GetName]"
			irv_announce = yes
		}
	}
}

# Tell the human players who know the rival that it has taken its ambition, once. Scope: rival.
irv_announce = {
	if = {
		limit = { NOT = { has_country_flag = irv_announced } }
		set_country_flag = irv_announced
		every_playable_country = {
			limit = {
				is_ai = no
				has_communications = prev
			}
			country_event = { id = irv.22 scopes = { from = prev } }
		}
	}
}

# --- Ideology ---------------------------------------------------------------------------------

# Found the rival's ideology from its primary ethic (vanilla generate_ideology_from_faction_or_ethic,
# common/scripted_effects/forceofwill_scripted_effects.txt). Scope: rival.
# Inlined per ethic, no helper effect: the vanilla chain is 3 levels deep (from_faction_or_ethic →
# from_faction → generate_ideology), and scripted effects stop running beyond 5 (modding insights,
# doc 07). This way irv.10 → here → vanilla stays at 4.
irv_found_ideology = {
	if = {
		limit = { has_country_flag = irv_primary_egalitarian NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = egalitarian }
	}
	if = {
		limit = { has_country_flag = irv_primary_authoritarian NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = authoritarian }
	}
	if = {
		limit = { has_country_flag = irv_primary_xenophile NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = xenophile }
	}
	if = {
		limit = { has_country_flag = irv_primary_xenophobe NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = xenophobe }
	}
	if = {
		limit = { has_country_flag = irv_primary_pacifist NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = pacifist }
	}
	if = {
		limit = { has_country_flag = irv_primary_militarist NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = militarist }
	}
	if = {
		limit = { has_country_flag = irv_primary_spiritualist NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = spiritualist }
	}
	if = {
		limit = { has_country_flag = irv_primary_materialist NOT = { exists = ideology } }
		generate_ideology_from_faction_or_ethic = { ETHIC = materialist }
	}
}

	if = {
		limit = { has_country_flag = irv_$F$ }
		log = "IRV   $F$"
	}
}
