# Ideological Rivals: scripted triggers

# An AI empire that may be made a rival at game start: a regular empire only. Scope: country.
# - Gestalts cannot hold a fanatic ethic.
# - Non-playable origins mark scripted empires (vanilla 01_origins_non_playable.txt: fallen
#   empires, pre-FTL, separatists, Khan successors, NPC hegemons, Slavers, ...).
# - Origins whose `possible` requires or forbids an ethic (00_origins.txt, 4.6.0): new ethics
#   would break them. change_government fails then (run B: Forever Cruise, fanatic xenophobe).
#   Endbringers are also a crisis already.
# - Lost Makers have their own ideology; Legendary Leader origins tie the authority.
irv_can_be_rival = {
	is_country_type = default
	is_ai = yes
	is_gestalt = no
	exists = capital_scope
	NOT = { has_country_flag = irv_rival }
	NOR = {
		# non-playable (scripted) empires
		has_origin = origin_fallen_empire
		has_origin = origin_fallen_empire_hive
		has_origin = origin_enlightened
		has_origin = origin_default_pre_ftl
		has_origin = origin_separatists
		has_origin = origin_liberated
		has_origin = origin_demonic_incursion
		has_origin = origin_khan_successor
		has_origin = origin_life_seeded_ai_only
		has_origin = origin_common_ground_npc
		has_origin = origin_hegemon_npc
		has_origin = origin_imperial_vassal_overlord
		has_origin = origin_slavers
		has_origin = origin_nomadic_purger
		has_origin = origin_nomadic_settled
		has_origin = origin_nomadic_subject
		# origins with ethics in `possible`
		has_origin = origin_sacred_path
		has_origin = origin_heirs_of_the_khan
		has_origin = origin_forever_cruise
		has_origin = origin_mindwardens
		has_origin = origin_endbringers
		has_origin = origin_cybernetic_creed
		has_origin = origin_synthetic_fertility
		has_origin = origin_broken_shackles
		has_origin = origin_shroudwalker_apprentice
		has_origin = origin_necrophage
		has_origin = origin_scion
		has_origin = origin_common_ground
		has_origin = origin_hegemon
		has_origin = origin_unplugged
		has_origin = origin_mechanists
		has_origin = origin_fruitful
		# own ideology, fixed authority
		has_origin = origin_lost_makers
		has_origin = origin_legendary_leader
		has_origin = origin_legendary_leader_imperial
		has_origin = origin_legendary_leader_dictatorial
	}
}

# A Force of Will rival that should found its ideology now. Scope: country. Crisis rivals never
# found one: Become the Crisis and Hyperthermia require not being a custodian.
# Mirrors vanilla can_generate_ideology's contact clause: an ideology needs a known empire.
irv_rival_needs_ideology = {
	has_country_flag = irv_aim_fow
	ideologies_enabled = yes
	NOT = { exists = ideology }
	any_playable_country = { has_communications = root }
}

# For irv_random_ethic. Scope: rival.
# A primary may not reuse an axis another rival holds; a secondary may not reuse its own primary's.
irv_axis_blocked_primary = { has_global_flag = irv_axis_$AXIS$ }
irv_axis_blocked_secondary = { has_country_flag = irv_axis_$AXIS$ }
irv_axis_blocked_fallback = { has_global_flag = irv_axis_$AXIS$ }

# For irv_random_ethic: an ethic the player holds is no opposition; a primary needs a reachable
# ambition. Scope: rival.
irv_ethic_barred_primary = {
	OR = {
		irv_player_holds = { E = $E$ }
		NOT = { has_global_flag = irv_reachable_$E$ }
	}
}
irv_ethic_barred_secondary = { irv_player_holds = { E = $E$ } }
irv_ethic_barred_fallback = { irv_player_holds = { E = $E$ } }
irv_player_holds = {
	event_target:irv_player = {
		OR = {
			has_ethic = ethic_$E$
			has_ethic = ethic_fanatic_$E$
		}
	}
}

# Does this game offer the ambition a fanatic $E$ rival would pursue? Militarist: Become the
# Crisis (Nemesis); xenophobe: Galactic Hyperthermia (Infernals); both barred on Civilian (vanilla
# blocks_disruptive_gameplay). Authoritarian: none (Force of Will bars it). Any other: Force of
# Will (Willpower, ideologies on). Read once into global flags by irv_init_reachable.
irv_reachable_militarist = { has_nemesis = yes blocks_disruptive_gameplay = no }
irv_reachable_xenophobe = { has_infernals = yes blocks_disruptive_gameplay = no }
irv_reachable_fow = { has_force_of_will_dlc = yes ideologies_enabled = yes }
